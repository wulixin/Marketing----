data['名义GDP增长率'] = data['国内生产总值(亿元)'].pct_change() * 100
data['CPI增长率'] = data['居民消费价格指数(上年=100)'] - 100
data['实际GDP增长率'] = data['名义GDP增长率'] - data['CPI增长率']
data['就业增长率'] = data['就业人员(万人)'].pct_change() * 100
data['就业弹性系数'] = data['就业增长率'] / data['名义GDP增长率']
data['人口增长率'] = data['年末总人口(万人)'].pct_change() * 100
data['人均GDP增长率'] = data['名义GDP增长率'] - data['人口增长率']
data['劳动年龄人口_估算'] = data['年末总人口(万人)'] * 0.4
data['劳动参与率'] = data['就业人员(万人)'] / data['劳动年龄人口_估算']
data['失业率_估算'] = (data['劳动年龄人口_估算'] - data['就业人员(万人)']) / data['劳动年龄人口_估算']
data['通胀调整失业率'] = data['失业率_估算'] * (1 + data['CPI增长率']/100)
# 计算汇率增长率（负值表示人民币贬值）
data['汇率增长率_美元'] = data['人民币对美元汇率(美元=100)(元)'].pct_change() * 100

核心结论

汇率增长率_美元 对实际GDP增长率的影响 不显著，但符号为负，表明 人民币升值可能对经济增长有正面作用。 结合今年特朗普上任以来，人民币持续增值，GDP5.3%跟生活常识相符合的。外资持续流入，港股和A股市场表现较好，人民币资产持续升值！

政策建议

汇率政策：适度保持人民币汇率稳定，避免大幅贬值或升值。

产业结构优化：继续推动工业化进程，提升工业占比，增强经济增长潜力。

进一步研究：补充进出口数据，构建更全面的模型，验证汇率变动对经济增长的具体机制。

维度 2：第二产业，第三产业就业弹性分析（增长带动就业的能力）¶

情况 含义 第三产业弹性 > 第二产业 服务业增长更能带动就业 → 支持“发展服务业稳就业”政策 弹性 < 0.1 “有增长无就业” → 增长未有效转化为就业 弹性 > 1 就业增长快于GDP增长 → 劳动密集型或初创阶段 弹性为负 GDP增长但就业下降 → 技术替代、去产能等
